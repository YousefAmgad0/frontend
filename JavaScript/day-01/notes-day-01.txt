(??) بيديك قيمة بديلة فقط لو القيمة الأصلية null أو undefined.
(??) cares about null and undefined
(||) cares about Falsy
مثال من API:
const user = {
  name: "Yousef",
  age: null
};

const age = user.age ?? 18;
console.log(age); // 18

Usage

1. API Data >>> const name = user.name ?? "Unknown";
2. Default values  >>> const limit = options.limit ?? 10;

فكره ال callback
انا معايا فانكشن بتعمل حاجه معينه وهديك فانكشن تاني عايزك تنفذها بعج مالفانكشن الاوله تخلص شغلها

function doSomething(callback) {
    // اعمل أي شغل
    console.log("Doing something...");

    // لما تخلص
    callback();
}

function showMessage() {
    console.log("Done!");
}
doSomething(showMessage);

function sendNotification() {
    console.log("Notification sent!");
}

doSomething(sendNotification);
